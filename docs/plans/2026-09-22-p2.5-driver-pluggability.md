**Goal:** 让 `storage`/`queue`/`dataops` 的第三方驱动成为**编译期可选项** —— 能力核心只留接口 + 注册表，驱动独立成包并由形态清单显式 import 注册；`full` 保留全部驱动（零退化），`enterprise` 收敛为 local + csv，`aws-sdk-go-v2`（67 个包）、`excelize`、`kafka-go`、`amqp091-go` 从轻形态的 import 闭包与二进制中消失。

**Architecture:** 每个驱动独立成包（`storage/{local,s3}`、`queue/{redis,kafka,rabbitmq}`、`dataops/{csv,excel}`），包内 `init()` 调用能力核心的 `Register`；能力核心删除 `switch`/硬编码注册，改为查注册表，未注册即 fail-closed 报错。声明分两层：`contract.Descriptor.Drivers`（能力支持的驱动包名，能力自描述）+ `assembly.Capability.Drivers`（该形态选中的子集），形态入口用 blank import 落实选中集，`tools/checkcapabilities` 静态校验「选中 == 闭包」。

**Tech Stack:** Go 1.26 · `golang.org/x/tools/go/packages`（闭包分析，`tools/composereport` 既有依赖）· testify

**Spec:** `docs/design/2026-09-18-capability-plugins-design.md` §3.7 / §6.3 / §9 / §11；本仓 `configs/*.yaml` 逐字节不变、catalog 仍 18 项、层②不减小 `go.mod`。

## 裁定（brainstorming 已确认，执行前必读）

1. **轻形态承载 = `enterprise` 收敛为 local + csv。** 5 个形态里只有 `full` 与 `enterprise` 含 `storage`+`dataops`；验收「只用本地存储/CSV 时重型依赖不出现」由 `enterprise` 兑现（aws 67 包 → 0、excelize 1 → 0）。**行为变更**：`enterprise` 下 `storage.type: s3|oss|minio` 与 xlsx 导入/导出改为 fail-closed 报错；`full` 不受影响（全部驱动都在）。
2. **注册机制 = 驱动包 `init()` 自注册 + 形态清单声明**（设计 §3.7 原文认可的形态：`init()` 只发生在被显式 import 的包内）。核心保留接口 + 注册表，`switch` 删除；注册表在包初始化期写入、启动后只读，不加锁；重复注册 panic。
3. **声明落点 = `contract.Descriptor.Drivers`（可用集）+ `assembly.Capability.Drivers`（选中集）**，门禁归 Go 工具（`tools/checkcapabilities`，入口 `make check-capabilities`）。**驱动名 = 驱动包目录名**，包路径 = `internal/capabilities/<cap>/<driver>`，因此门禁与 P2.7 脚手架都不需要对照表；一个驱动包可注册多个配置取值（`s3` 包覆盖 `s3`/`oss`/`minio`）。
4. **govulncheck 豁免保留。** 拆包**不解除** `GO-2026-6452` 的可达性（`govulncheck ./...` 扫整个 module，`full` 仍 import Excel 驱动），只把 `scripts/govulncheck.sh` 注释里的复核条件改为「excelize ≥ v2.11.1」并在 release note 说明。
5. **沿用硬约束**：`configs/*.yaml` 逐字节不变；catalog 仍 18 项；层②不减小 `go.mod`；能力间只经 `contract`；`internal/kernel/**` 不 import `internal/capabilities/**`；`internal/app` 不 import `catalog`/能力根包/`assembly`（三个 domain 叶子包偏差见 P2.4 裁定 13）；**禁止自动提交**（AGENTS.md 最高优先级，每个 Task 的 commit 仅在用户明确指令后执行）。

```go
package storage

import (
	"fmt"
	"sort"
	"strings"
)

// Factory 按配置构造存储实现。驱动包在 init() 中调用 Register 注册。
type Factory func(Config) (Storage, error)

// drivers 是本构建已注册的驱动表。写入只发生在包初始化期（Go 保证 init 串行且先于
// main），启动后只读，因此不加锁。
var drivers = map[StorageType]Factory{}

// Register 注册存储驱动，仅供驱动包在 init() 中调用。重复注册是编码错误，直接 panic
// （与 database/sql.Register 同形）。
func Register(t StorageType, f Factory) {
	if _, dup := drivers[t]; dup {
		panic("storage: driver already registered: " + string(t))
	}
	drivers[t] = f
}

// RegisteredTypes 返回本构建已注册的存储类型（升序），用于 fail-closed 文案与诊断。
func RegisteredTypes() []StorageType {
	out := make([]StorageType, 0, len(drivers))
	for t := range drivers {
		out = append(out, t)
	}
	sort.Slice(out, func(i, j int) bool { return out[i] < out[j] })
	return out
}

// New 按配置创建存储实例：空类型按 local 处理（与下沉前一致）。配置的类型未编译进
// 本构建时明确报错，不静默回退到其它驱动。
func New(cfg Config) (Storage, error) {
	if cfg.Type == "" {
		cfg.Type = StorageTypeLocal
	}
	f, ok := drivers[cfg.Type]
	if !ok {
		return nil, fmt.Errorf("storage driver %q is not compiled into this build (compiled: %s)",
			cfg.Type, typesList(RegisteredTypes()))
	}
	return f(cfg)
}

// typesList 渲染已注册类型清单；空集渲染为 none。
func typesList(types []StorageType) string {
	if len(types) == 0 {
		return "none"
	}
	out := make([]string, len(types))
	for i, t := range types {
		out[i] = string(t)
	}
	return strings.Join(out, ", ")
}
```

```go
package queue

import (
	"errors"
	"fmt"
	"sort"
	"strings"
)

// Factory 按配置构造队列实现。驱动包在 init() 中调用 Register 注册。
type Factory func(Config) (Queue, error)

// ErrUnregisteredDriver 标记「配置的类型未编译进本构建」，供调用方 errors.Is 判定。
var ErrUnregisteredDriver = errors.New("queue driver is not compiled into this build")

// drivers 是本构建已注册的驱动表（init 期写入，启动后只读，不加锁）。
var drivers = map[Type]Factory{}

// Register 注册队列驱动，仅供驱动包在 init() 中调用；重复注册 panic。
func Register(t Type, f Factory) {
	if _, dup := drivers[t]; dup {
		panic("queue: driver already registered: " + string(t))
	}
	drivers[t] = f
}

// RegisteredTypes 返回本构建已注册的队列类型（升序）。
func RegisteredTypes() []Type {
	out := make([]Type, 0, len(drivers))
	for t := range drivers {
		out = append(out, t)
	}
	sort.Slice(out, func(i, j int) bool { return out[i] < out[j] })
	return out
}

// EnsureRegistered 校验配置的队列类型已编译进本构建（queue.Wire 在启动时调用）。
func EnsureRegistered(t Type) error {
	if _, ok := drivers[t]; !ok {
		return fmt.Errorf("queue driver %q is not compiled into this build (compiled: %s): %w",
			t, typesList(RegisteredTypes()), ErrUnregisteredDriver)
	}
	return nil
}

// New 按类型创建队列；类型未编译进本构建时明确报错，不静默回退。
func New(cfg Config) (Queue, error) {
	if err := EnsureRegistered(cfg.Type); err != nil {
		return nil, err
	}
	return drivers[cfg.Type](cfg)
}

// typesList 渲染已注册类型清单；空集渲染为 none。
func typesList(types []Type) string {
	if len(types) == 0 {
		return "none"
	}
	out := make([]string, len(types))
	for i, t := range types {
		out[i] = string(t)
	}
	return strings.Join(out, ", ")
}
```

```go
// ImporterFactory 构造一个格式实现；驱动包在 init() 中注册。
type ImporterFactory func() Importer

// Registry 按格式管理导入器工厂。
type Registry struct{ factories map[Format]ImporterFactory }

func NewRegistry() *Registry { return &Registry{factories: make(map[Format]ImporterFactory)} }

// Register 注册格式实现；重复注册 panic（与 database/sql.Register 同形）。
func (r *Registry) Register(format Format, f ImporterFactory) {
	if _, dup := r.factories[format]; dup {
		panic("importer: format already registered: " + string(format))
	}
	r.factories[format] = f
}

// Get 返回格式实现；未注册时明确报错。
func (r *Registry) Get(format Format) (Importer, error) {
	f, ok := r.factories[format]
	if !ok {
		return nil, fmt.Errorf("unsupported import format: %s", format)
	}
	return f(), nil
}

// defaultRegistry 是进程级注册表：驱动包 init() 写入，启动后只读。
var defaultRegistry = NewRegistry()

// Register 把格式实现注册进进程级注册表（仅供驱动包 init() 调用）。
func Register(format Format, f ImporterFactory) { defaultRegistry.Register(format, f) }

// Get 从进程级注册表取格式实现；格式未编译进本构建时明确报错，不静默回退。
func Get(format Format) (Importer, error) {
	imp, err := defaultRegistry.Get(format)
	if err != nil {
		return nil, fmt.Errorf("%w (compiled: %s)", err, formatsList(RegisteredFormats()))
	}
	return imp, nil
}

// Supported 报告本构建是否注册了该格式。
func Supported(format Format) bool { _, ok := defaultRegistry.factories[format]; return ok }

// RegisteredFormats 返回本构建已注册的导入格式（升序）。
func RegisteredFormats() []Format {
	out := make([]Format, 0, len(defaultRegistry.factories))
	for f := range defaultRegistry.factories {
		out = append(out, f)
	}
	sort.Slice(out, func(i, j int) bool { return out[i] < out[j] })
	return out
}

// formatsList 渲染已注册格式清单；空集渲染为 none（错误文案与诊断共用）。
func formatsList(formats []Format) string {
	if len(formats) == 0 {
		return "none"
	}
	out := make([]string, len(formats))
	for i, f := range formats {
		out[i] = string(f)
	}
	return strings.Join(out, ", ")
}
```
