# v0.4.0 与 jimu-web 首版联动（后端）Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 把 jimu 后端接口与契约补到「jimu-web 能做出第一个真实管理页（`/platform/tenants`）并在真实后端上跑通」的程度，并让契约同步可验证。

**Architecture:** 全部改动落在既有能力内，不新增能力/表/迁移/形态。契约缺陷（两处重复 `basePath` 前缀）就地修；身份与权限经新增的 `GET /auth/me` 下发（auth 的**认证-only**受保护子组，权限点复用 `kernel/access` 的既有查询）；注册模式经新增的公开 `GET /auth/config` 对客户端可见；租户列表加 `keyword`/`status` 筛选；本轮消费端点的响应体由 `{}` 换成具体 DTO。

**Tech Stack:** Go 1.2x、Gin、GORM（MySQL/MariaDB/PostgreSQL/SQLite）、Casbin v3、swag（Swagger 2.0）、goose 迁移、testify、Vitest/Playwright（前端不在本仓）。

**Spec:** [`docs/superpowers/specs/2026-10-13-v0.4.0-portal-integration-design.md`](../specs/2026-10-13-v0.4.0-portal-integration-design.md)

```go
package interfaces

import (
	"jimu/internal/shared/response"

	"github.com/gin-gonic/gin"
)

// AuthConfigResponse 客户端可见的认证能力开关。
// 只暴露渲染注册入口所必需的项：是否需要显示注册入口、
// 以及注册是否等于「开通新租户」（决定 tenant_name 是否必填）。
type AuthConfigResponse struct {
	// PublicRegistration 是否开放公开注册（false 时 /auth/register 路由不注册）
	PublicRegistration bool `json:"public_registration"`
	// ProvisioningEnabled 开通式注册：注册 = 开通新租户，tenant_name 必填
	ProvisioningEnabled bool `json:"provisioning_enabled"`
}

// AuthConfig godoc
// @Summary      获取认证能力开关
// @Description  返回客户端渲染登录/注册入口所需的公开开关。无论是否开放公开注册，本端点都存在。
// @Tags         认证
// @Produce      json
// @Success      200  {object}  AuthConfigResponse  "成功，返回注册相关开关"
// @Router       /auth/config [get]
func (h *AuthHandler) AuthConfig(c *gin.Context) {
	response.OK(c, AuthConfigResponse{
		PublicRegistration:  h.cfg.PublicRegistration,
		ProvisioningEnabled: h.cfg.ProvisioningEnabled,
	})
}
```

```go
package application

import (
	"sort"

	"jimu/internal/kernel/access"
)

// PermissionPoint 一个有效权限点：与 Casbin 策略同形（resource 支持 keyMatch 通配）。
// 客户端按 (resource, action) 判定，不需要后端再翻译成权限码。
type PermissionPoint struct {
	Resource string `json:"resource"`
	Action   string `json:"action"`
}

// EffectivePermissions 由「用户角色」与「全量角色策略」推导有效权限点。
//
// 语义与 kernel/access 的判定保持一致：用户命中某角色即拥有该角色的全部策略点；
// 去重后按 (resource, action) 排序，保证同一输入永远得到同一顺序（便于断言与 diff）。
// 无有效权限时返回**空切片而非 nil**：响应的 permissions 必须是 [] 而不是 null。
func EffectivePermissions(roles []string, policies []access.Policy) []PermissionPoint {
	owned := make(map[string]bool, len(roles))
	for _, role := range roles {
		owned[role] = true
	}

	seen := make(map[PermissionPoint]bool, len(policies))
	out := make([]PermissionPoint, 0, len(policies))
	for _, policy := range policies {
		if !owned[policy.Role] {
			continue
		}
		point := PermissionPoint{Resource: policy.Resource, Action: policy.Action}
		if seen[point] {
			continue
		}
		seen[point] = true
		out = append(out, point)
	}

	sort.Slice(out, func(i, j int) bool {
		if out[i].Resource != out[j].Resource {
			return out[i].Resource < out[j].Resource
		}
		return out[i].Action < out[j].Action
	})
	return out
}
```

```go
package application

import (
	"context"
	"time"

	"jimu/internal/kernel/access"
	"jimu/internal/shared/errors"
)

// MeUser 当前会话用户的只读视图。
type MeUser struct {
	ID        uint64    `json:"id"`
	Username  string    `json:"username"`
	Status    int8      `json:"status"`
	TenantID  uint64    `json:"tenant_id"`
	CreatedAt time.Time `json:"created_at"`
}

// MeView 会话引导所需的全部信息：身份 + 角色（仅供展示）+ 有效权限点（用于判定）。
type MeView struct {
	User        MeUser            `json:"user"`
	Roles       []string          `json:"roles"`
	Permissions []PermissionPoint `json:"permissions"`
}

// Me 返回当前用户身份与有效权限点。
//
// tenantID 来自已验签 claim（gin context，由 auth.AuthMiddleware 写入），
// 不取账号行里的 tenant_id —— 后者可能滞后于签发时的归属。
// 权限点经 kernel/access 的既有查询（角色 + 策略）由 EffectivePermissions 聚合。
func (s *AuthService) Me(ctx context.Context, userID, tenantID uint64) (*MeView, error) {
	account, err := s.userRepo.FindByID(ctx, userID)
	if err != nil {
		return nil, errors.Wrap(errors.CodeUserNotFound, "user not found", err)
	}

	// 两个切片都初始化为空切片：响应的 roles/permissions 必须是 [] 而不是 null
	roles := []string{}
	policies := []access.Policy{}
	if s.authz != nil {
		if roles, err = s.authz.RolesForUser(ctx, userID); err != nil {
			return nil, errors.Wrap(errors.CodeInternalError, "failed to load roles", err)
		}
		if roles == nil {
			roles = []string{}
		}
		if policies, err = s.authz.Policies(ctx); err != nil {
			return nil, errors.Wrap(errors.CodeInternalError, "failed to load policies", err)
		}
	}

	return &MeView{
		User: MeUser{
			ID:        account.ID,
			Username:  account.Username,
			Status:    account.Status,
			TenantID:  tenantID,
			CreatedAt: account.CreatedAt,
		},
		Roles:       roles,
		Permissions: EffectivePermissions(roles, policies),
	}, nil
}
```

```go
	if !ok {

	result, _ := value.(uint64)
	return result

// @Summary      获取当前会话身份与权限点
// @Description  返回当前用户的身份、角色与有效权限点（resource + action，与 Casbin 策略同形，resource 支持通配）。

// @Failure      401  {object}  contract.ErrorResponse  "未认证"

		return

	response.OK(c, view)
```

```go
// Package sqlpattern 提供跨数据库一致的 SQL LIKE 模式构造。
package sqlpattern

import "strings"

// EscapeChar LIKE 的转义字符。用 '!' 而不是反斜杠：MySQL 字符串字面量里
// 反斜杠本身是转义字符，跨方言写起来易错；'!' 在 MySQL/PostgreSQL/SQLite 中
// 都是普通字面量，无需额外转义。
const EscapeChar = "!"

// EscapeLike 转义 LIKE 通配符，配合 ESCAPE '!' 使用。
func EscapeLike(s string) string {
	replacer := strings.NewReplacer(
		EscapeChar, EscapeChar+EscapeChar,
		"%", EscapeChar+"%",
		"_", EscapeChar+"_",
	)
	return replacer.Replace(s)
}

// LikeContains 生成子串匹配的 LIKE 模式（调用方仍需显式写 ESCAPE '!'）。
func LikeContains(query string) string {
	return "%" + EscapeLike(query) + "%"
}
```

```go
// NoAccountLogger 启动自检所需的最小日志面（*logger.Logger 满足）。
type NoAccountLogger interface {
	Errorw(msg string, keysAndValues ...interface{})
}

// WarnIfNoAccounts 是结构性种子被跳过时的最后一道自检：
// 库中一个账号都没有意味着 API 没有任何可用登录（生产关闭公开注册时即为死局），
// 此时给出 error 级提示与下一步；返回是否已提示。
//
// 表不存在或查询失败一律静默降级：容器启动早于 CLI 迁移是已知且允许的时序，
// 服务启动不得因为缺 ADMIN_PASSWORD 或表未建而失败。
func WarnIfNoAccounts(db *gorm.DB, log NoAccountLogger) bool {
	if db == nil || log == nil {
		return false
	}
	var count int64
	if err := db.Model(&userdomain.User{}).Count(&count).Error; err != nil {
		return false
	}
	if count > 0 {
		return false
	}
	log.Errorw("no account exists and structural seed is disabled; the API has no usable login",
		"hint", "set ADMIN_PASSWORD and restart the server, or enable auth.public_registration")
	return true
}
```
