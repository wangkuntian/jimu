# Captcha Verification Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增图形验证码服务，登录/注册防刷。生成验证码存 Redis，登录/注册请求带 `captcha_id` + `captcha_code` 服务端校验。

**Architecture:** `internal/platform/captcha/` 提供生成与校验服务（`base64Captcha` + Redis 存储）。auth 模块 `New` 时注入 captcha 服务，`Login`/`Register` handler 内校验（非中间件，便于配置开关）。新路由 `GET /api/v1/captcha` 生成验证码。

**Tech Stack:** Go 1.26.5, `github.com/mojocn/base64Captcha`, 现有 `go-redis/v9`

## Global Constraints

- 模块：`jimu`
- 新依赖：`github.com/mojocn/base64Captcha`
- Redis key 格式：`jimu:captcha:{id}`，TTL 5 分钟
- 校验失败返回统一错误码：`4xxx` 验证码模块
- 遵循现有代码风格：中文注释、swagger 中文注解、统一响应
- 保持向后兼容：`captcha.enabled=false` 时登录/注册不校验验证码

// internal/platform/captcha/captcha.go
package captcha

import (
	"context"
	"fmt"
	"time"

	"github.com/mojocn/base64Captcha"
	"github.com/redis/go-redis/v9"
)

// Service 验证码服务（生成 + Redis 存储 + 校验）
type Service struct {
	client *redis.Client
	store  *base64Captcha.RedisStore
	ttl    time.Duration
}

// NewService 创建验证码服务
func NewService(client *redis.Client, ttl time.Duration) *Service {
	return &Service{
		client: client,
		store:  base64Captcha.NewDefaultRedisStoreWithDB(client, 0, ttl),
		ttl:    ttl,
	}
}

// Generate 生成验证码，返回 id 与 base64 图片
func (s *Service) Generate(ctx context.Context) (string, string, error) {
	driver := base64Captcha.NewDriverDigit(80, 240, 4, 0.7, 80)
	captcha := base64Captcha.NewCaptcha(driver, s.store)
	id, b64, _, err := captcha.Generate()
	if err != nil {
		return "", "", fmt.Errorf("generate captcha: %w", err)
	}
	return id, b64, nil
}

// Verify 校验验证码，校验后无论成败都删除（一次性）
func (s *Service) Verify(ctx context.Context, id, code string) error {
	key := fmt.Sprintf("jimu:captcha:%s", id)
	stored, err := s.client.Get(ctx, key).Result()
	if err != nil {
		return fmt.Errorf("captcha not found or expired")
	}
	_ = s.client.Del(ctx, key).Err()
	if stored != code {
		return fmt.Errorf("captcha code mismatch")
	}
	return nil
}
```

// internal/platform/captcha/captcha_test.go
package captcha

import (
	"context"
	"testing"
	"time"

	"github.com/alicebob/miniredis/v2"
	"github.com/redis/go-redis/v9"
	"github.com/stretchr/testify/assert"
)

func TestGenerate(t *testing.T) {
	mr := miniredis.RunT(t)
	client := redis.NewClient(&redis.Options{Addr: mr.Addr()})
	s := NewService(client, 5*time.Minute)

	id, img, err := s.Generate(context.Background())
	assert.NoError(t, err)
	assert.NotEmpty(t, id)
	assert.Contains(t, img, "data:image")
}

func TestVerify(t *testing.T) {
	mr := miniredis.RunT(t)
	client := redis.NewClient(&redis.Options{Addr: mr.Addr()})
	s := NewService(client, 5*time.Minute)

	// 直接写入一个验证码验证校验逻辑
	key := "jimu:captcha:test123"
	_ = client.Set(context.Background(), key, "1234", 5*time.Minute).Err()

	err := s.Verify(context.Background(), "test123", "1234")
	assert.NoError(t, err)

	// 一次性：再次校验失败
	err = s.Verify(context.Background(), "test123", "1234")
	assert.Error(t, err)
}
```

// internal/platform/captcha/handler.go
package captcha

import (
	"jimu/internal/shared/response"

	"github.com/gin-gonic/gin"
)

// CaptchaHandler 验证码 HTTP 处理器
type CaptchaHandler struct {
	service *Service
}

// NewCaptchaHandler 创建验证码处理器
func NewCaptchaHandler(service *Service) *CaptchaHandler {
	return &CaptchaHandler{service: service}
}

// Generate godoc
// @Summary      获取验证码
// @Description  生成图形验证码，返回 captcha_id 与 base64 图片
// @Tags         验证码
// @Produce      json
// @Success      200  {object}  response.Body  "成功，返回 captcha_id 和 captcha_image"
// @Router       /captcha [get]
func (h *CaptchaHandler) Generate(c *gin.Context) {
	id, img, err := h.service.Generate(c.Request.Context())
	if err != nil {
		response.Fail(c, err)
		return
	}
	response.OK(c, gin.H{
		"captcha_id":    id,
		"captcha_image": img,
	})
}
```

func RegisterAuthRoutes(r *gin.RouterGroup, service *application.AuthService, jwtUtil *auth.JWT, cfg config.AuthConfig, limiter *auth.Limiter, captchaSvc *captcha.Service) {
	handler := NewAuthHandler(service, cfg, limiter, captchaSvc)
	authGroup := r.Group("/auth")
	{
		authGroup.POST("/login", middleware.ValidateJSON(&loginRequest{}), handler.Login)
		if cfg.PublicRegistration {
			authGroup.POST("/register", middleware.ValidateJSON(&loginRequest{}), handler.Register)
		}
		authGroup.POST("/refresh", middleware.ValidateJSON(&refreshRequest{}), handler.RefreshToken)

		protected := authGroup.Group("")
		protected.Use(auth.AuthMiddleware(jwtUtil))
		protected.POST("/logout", handler.Logout)
		protected.POST("/logout-all", handler.LogoutAll)
	}
}

// RegisterCaptchaRoute 注册验证码路由
func RegisterCaptchaRoute(r *gin.RouterGroup, svc *captcha.Service) {
	if svc == nil {
		return
	}
	handler := captcha.NewCaptchaHandler(svc)
	r.GET("/captcha", handler.Generate)
}
```

- [ ] **Step 6: 修改 auth handler.go**

`AuthHandler` 加 `captcha *captcha.Service` 与 `cfg config.CaptchaConfig`，`NewAuthHandler` 加参数。`Login`/`Register` 开头加：

```go
func (h *AuthHandler) verifyCaptcha(c *gin.Context, req *loginRequest) bool {
	if h.cfg.Enabled {
		if req.CaptchaID == "" || req.CaptchaCode == "" {
			response.Fail(c, errors.New(errors.CodeCaptchaRequired, "captcha required"))
			return false
		}
		if err := h.captcha.Verify(c.Request.Context(), req.CaptchaID, req.CaptchaCode); err != nil {
			response.Fail(c, errors.New(errors.CodeCaptchaInvalid, "invalid captcha"))
			return false
		}
	}
	return true
}
```

在 `Login` 限流检查前调用 `if !h.verifyCaptcha(c, req) { return }`。
